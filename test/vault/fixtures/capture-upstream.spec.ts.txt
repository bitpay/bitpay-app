import getStore, {storage} from './index';
import {flushPersistor} from './persistor';
import {migrateWalletSecrets} from './wallet-secrets/wallet-secrets.effects';
import {removePersistRootBackups, resumePersistRootBackups} from './backup/fs-backup';
import {initialWalletSecretsState} from './wallet-secrets/wallet-secrets.reducer';

it('captures the real first and second upstream persistence flushes', async () => {
  const {store, persistor} = await getStore();
  await new Promise<void>(resolve => {
    if (persistor.getState().bootstrapped) return resolve();
    const unsubscribe = persistor.subscribe(() => {
      if (persistor.getState().bootstrapped) { unsubscribe(); resolve(); }
    });
  });
  const key = {
    id: 'fixture',
    properties: {mnemonic: 'synthetic upstream mnemonic', xPrivKey: 'synthetic upstream xpriv'},
    wallets: [{id: 'wallet-fixture', credentials: {
      xPubKey: 'synthetic-public', requestPrivKey: 'synthetic-request-private',
      walletPrivKey: 'synthetic-wallet-private', personalEncryptingKey: 'synthetic-personal',
      sharedEncryptingKey: 'synthetic-shared', mnemonic: 'synthetic-credential-mnemonic',
      mnemonicEncrypted: 'synthetic-inner-password-ciphertext', entropySource: 'synthetic-entropy',
    }}],
    tssSession: {id: 'tss-fixture', partyKey: {xPrivKey: 'synthetic-tss-private'}, sessionExport: 'synthetic-session-export', password: 'synthetic-ceremony-password', copayers: [{partyId: 1, joinCode: 'synthetic-join-code'}], status: 'ceremony_in_progress'},
  };
  const pendingJoinerSession = {sessionId: 'pending-fixture', partyKey: {xPrivKey: 'synthetic-pending-private'}, createdAt: 1};
  store.dispatch({type: 'persist/REHYDRATE', key: 'root', payload: {
    APP: {...store.getState().APP, migrationMMKVStorageComplete: true},
    WALLET: {...store.getState().WALLET, keys: {fixture: key}, pendingJoinerSession, secretsMigrated: false},
    WALLET_SECRETS: initialWalletSecretsState,
  }});
  const snapshots: string[] = [];
  await store.dispatch(migrateWalletSecrets(async () => {
    await flushPersistor();
    snapshots.push(storage.getString('persist:root')!);
  }, removePersistRootBackups, resumePersistRootBackups) as any);
  expect(snapshots).toHaveLength(2);
  for (const [i, raw] of snapshots.entries()) {
    const root = JSON.parse(raw);
    const wallet = JSON.parse(JSON.parse(root.WALLET));
    expect(wallet.secretsMigrated).toBe(i === 1);
    expect(JSON.parse(root.WALLET_SECRETS).startsWith('U2FsdGVkX1')).toBe(true);
    expect(Boolean(wallet.keys.fixture.properties)).toBe(i === 0);
  }
  require('fs').writeFileSync(process.env.BIP02_CAPTURE_PATH!, JSON.stringify({
    sourceCommit: '6c84ecc4c0bee99d599449a29f2c6447f451478a',
    oldKey: 'synthetic-upstream-key', key, pendingJoinerSession,
    intermediate: snapshots[0], completed: snapshots[1],
  }, null, 2) + '\n');
  persistor.pause();
});
